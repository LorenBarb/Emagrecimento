    <script>
    document.addEventListener('DOMContentLoaded', () => {
        // Inicializa os ícones
        lucide.createIcons();
        
        // --- ESTADO DA APLICAÇÃO ---
        let state = {
            profile: {
                name: 'Usuário',
                age: 30,
                height: 175, // em cm
                gender: 'male' // 'male' ou 'female'
            },
            logs: []
        };
        
        // --- ELEMENTOS DO DOM ---
        const progressChartCtx = document.getElementById('progressChart').getContext('2d');
        const chartFilter = document.getElementById('chartFilter');
        const metricsGrid = document.getElementById('metrics-grid');
        const historyList = document.getElementById('history-list');
        const logForm = document.getElementById('logForm');
        const profileForm = document.getElementById('profileForm');

        let progressChart;

        // --- DEFINIÇÕES E CÁLCULOS ---
        
        // Nomes amigáveis para as chaves de métricas
        const metricLabels = {
            peso: 'Peso',
            imc: 'IMC',
            gordura_corporal: 'Gordura Corporal',
            taxa_muscular: 'Taxa Muscular',
            massa_livre_gordura: 'Massa Livre de Gordura',
            gordura_subcutanea: 'Gordura Subcutânea',
            gordura_visceral: 'Gordura Visceral',
            agua_corporal: 'Água Corporal',
            massa_muscular: 'Massa Muscular',
            massa_ossea: 'Massa Óssea',
            proteina: 'Proteína',
            tmb: 'TMB',
            idade_metabolica: 'Idade Metabólica',
            medidas_corporais: 'Medidas Corporais'
        };

        // Função para obter os valores ideais.
        // Estes são valores de referência e podem variar.
        const getIdealRange = (metric, value) => {
            const { gender, age, height } = state.profile;
            value = parseFloat(value);
            
            // Status: 'ideal', 'bom', 'atencao', 'alto', 'baixo'
            switch (metric) {
                case 'imc':
                    if (value < 18.5) return { range: 'Abaixo de 18.5', status: 'baixo' };
                    if (value <= 24.9) return { range: '18.5 - 24.9', status: 'ideal' };
                    if (value <= 29.9) return { range: '25.0 - 29.9', status: 'atencao' };
                    return { range: 'Acima de 30.0', status: 'alto' };
                case 'gordura_corporal':
                    if (gender === 'male') {
                        if (age <= 39) {
                            if (value < 8) return { range: '8-20%', status: 'baixo' };
                            if (value <= 20) return { range: '8-20%', status: 'ideal' };
                            return { range: '8-20%', status: 'alto' };
                        } else {
                            if (value < 11) return { range: '11-22%', status: 'baixo' };
                            if (value <= 22) return { range: '11-22%', status: 'ideal' };
                            return { range: '11-22%', status: 'alto' };
                        }
                    } else { // female
                        if (age <= 39) {
                            if (value < 21) return { range: '21-33%', status: 'baixo' };
                            if (value <= 33) return { range: '21-33%', status: 'ideal' };
                            return { range: '21-33%', status: 'alto' };
                        } else {
                            if (value < 23) return { range: '23-34%', status: 'baixo' };
                            if (value <= 34) return { range: '23-34%', status: 'ideal' };
                            return { range: '23-34%', status: 'alto' };
                        }
                    }
                case 'gordura_visceral':
                     if (value <= 9) return { range: '1 - 9', status: 'ideal' };
                     if (value <= 14) return { range: '1 - 9', status: 'atencao' };
                     return { range: '1 - 9', status: 'alto' };
                case 'agua_corporal':
                    const idealWater = gender === 'male' ? 60 : 55;
                    if (value >= idealWater - 5 && value <= idealWater + 5) return { range: `${idealWater-5}-${idealWater+5}%`, status: 'ideal' };
                    return { range: `${idealWater-5}-${idealWater+5}%`, status: 'atencao' };
                case 'taxa_muscular':
                    if (gender === 'male') {
                        if (value >= 33.3) return { range: '> 33.3%', status: 'ideal' };
                        return { range: '> 33.3%', status: 'baixo' };
                    } else { // female
                         if (value >= 27.3) return { range: '> 27.3%', status: 'ideal' };
                         return { range: '> 27.3%', status: 'baixo' };
                    }
                case 'massa_ossea':
                    // Varia muito com peso corporal, simplificado aqui.
                    const lastLog = getLatestLog();
                    if (!lastLog || !lastLog.peso) return { range: 'N/A', status: 'atencao' };
                    let idealBoneMass;
                    if (gender === 'male') {
                        if (lastLog.peso < 65) idealBoneMass = 2.65;
                        else if (lastLog.peso <= 95) idealBoneMass = 3.29;
                        else idealBoneMass = 3.69;
                    } else {
                        if (lastLog.peso < 50) idealBoneMass = 1.95;
                        else if (lastLog.peso <= 75) idealBoneMass = 2.40;
                        else idealBoneMass = 2.95;
                    }
                    if (Math.abs(value - idealBoneMass) < 0.5) return { range: `~${idealBoneMass.toFixed(2)} kg`, status: 'ideal' };
                    return { range: `~${idealBoneMass.toFixed(2)} kg`, status: 'atencao' };
                default:
                    return { range: 'N/A', status: 'neutro' };
            }
        };
        
        const getStatusColor = (status) => {
            switch(status) {
                case 'ideal': return 'bg-green-500';
                case 'bom': return 'bg-blue-500';
                case 'atencao': return 'bg-yellow-500';
                case 'alto':
                case 'baixo': 
                    return 'bg-red-500';
                default: return 'bg-gray-400';
            }
        };

        // --- FUNÇÕES DE RENDERIZAÇÃO ---
        
        const renderMetrics = () => {
            metricsGrid.innerHTML = '';
            const lastLog = getLatestLog();

            if (!lastLog) {
                metricsGrid.innerHTML = `<p class="col-span-full text-center text-gray-500">Adicione um registro para ver suas métricas.</p>`;
                return;
            }

            // Calculos derivados
            if (state.profile.height && lastLog.peso) {
                const heightInMeters = state.profile.height / 100;
                lastLog.imc = (lastLog.peso / (heightInMeters * heightInMeters)).toFixed(2);
            }
            if (lastLog.peso && lastLog.gordura_corporal) {
                lastLog.massa_livre_gordura = (lastLog.peso * (1 - lastLog.gordura_corporal / 100)).toFixed(2);
            }

            const metricsToDisplay = [
                'peso', 'imc', 'gordura_corporal', 'taxa_muscular', 'gordura_visceral', 'agua_corporal',
                'massa_muscular', 'massa_ossea', 'proteina', 'tmb', 'idade_metabolica', 'massa_livre_gordura'
            ];
            
            metricsToDisplay.forEach(key => {
                if (lastLog[key] !== undefined && lastLog[key] !== null) {
                    const value = lastLog[key];
                    const unit = getUnit(key);
                    const { range, status } = getIdealRange(key, value);
                    
                    const card = `
                        <div class="metric-card bg-gray-50 p-4 rounded-lg flex flex-col justify-between">
                            <div>
                                <div class="flex items-center justify-between text-sm text-gray-500">
                                    <span>${metricLabels[key] || key}</span>
                                    <span class="h-2.5 w-2.5 rounded-full ${getStatusColor(status)}"></span>
                                </div>
                                <p class="text-2xl font-bold mt-1">${value} <span class="text-lg font-medium">${unit}</span></p>
                            </div>
                            <p class="text-xs text-gray-400 mt-2">Ideal: ${range}</p>
                        </div>
                    `;
                    metricsGrid.innerHTML += card;
                }
            });
        };
        
        const getUnit = (metric) => {
             const units = {
                peso: 'kg',
                gordura_corporal: '%',
                taxa_muscular: '%',
                gordura_visceral: '',
                agua_corporal: '%',
                massa_muscular: 'kg',
                massa_ossea: 'kg',
                proteina: '%',
                tmb: 'kcal',
                idade_metabolica: 'anos',
                massa_livre_gordura: 'kg',
                imc: ''
            };
            return units[metric] || '';
        }

        const renderChart = () => {
            const days = parseInt(chartFilter.value);
            const filteredLogs = state.logs
                .map(log => ({ ...log, dateObj: new Date(log.date) }))
                .sort((a, b) => a.dateObj - b.dateObj)
                .slice(-days);

            const labels = filteredLogs.map(log => new Date(log.date).toLocaleDateString('pt-BR', { day: '2-digit', month: '2-digit' }));
            const weightData = filteredLogs.map(log => log.peso);
            const fatData = filteredLogs.map(log => log.gordura_corporal);

            if (progressChart) {
                progressChart.destroy();
            }

            progressChart = new Chart(progressChartCtx, {
                type: 'line',
                data: {
                    labels: labels,
                    datasets: [{
                        label: 'Peso (kg)',
                        data: weightData,
                        borderColor: 'rgb(59, 130, 246)',
                        backgroundColor: 'rgba(59, 130, 246, 0.1)',
                        fill: true,
                        tension: 0.3,
                        yAxisID: 'y'
                    }, {
                        label: 'Gordura Corporal (%)',
                        data: fatData,
                        borderColor: 'rgb(239, 68, 68)',
                        backgroundColor: 'rgba(239, 68, 68, 0.1)',
                        fill: true,
                        tension: 0.3,
                        yAxisID: 'y1'
                    }]
                },
                options: {
                    responsive: true,
                    maintainAspectRatio: false,
                    scales: {
                        y: {
                            type: 'linear',
                            display: true,
                            position: 'left',
                            title: {
                                display: true,
                                text: 'Peso (kg)'
                            }
                        },
                        y1: {
                            type: 'linear',
                            display: true,
                            position: 'right',
                             title: {
                                display: true,
                                text: 'Gordura Corporal (%)'
                            },
                            grid: {
                                drawOnChartArea: false, 
                            },
                        },
                    }
                }
            });
        };
        
        const renderHistory = () => {
            historyList.innerHTML = '';
            if (state.logs.length === 0) {
                 historyList.innerHTML = `<p class="text-gray-500 text-center">Nenhum registro encontrado.</p>`;
                 return;
            }
            
            const sortedLogs = [...state.logs].sort((a, b) => new Date(b.date) - new Date(a.date));
            sortedLogs.forEach(log => {
                const date = new Date(log.date);
                const item = `
                    <div class="flex justify-between items-center p-2 rounded-md hover:bg-gray-100">
                        <div>
                            <p class="font-semibold">${date.toLocaleDateString('pt-BR', {weekday: 'long', day: '2-digit', month: 'short'})}</p>
                            <p class="text-sm text-gray-500">${log.peso} kg • ${log.gordura_corporal || 'N/A'}% gordura</p>
                        </div>
                         <button onclick="deleteLog('${log.id}')" class="text-red-500 hover:text-red-700 p-1"><i data-lucide="trash-2" class="h-4 w-4"></i></button>
                    </div>
                `;
                historyList.innerHTML += item;
            });
            lucide.createIcons(); // Recriar ícones
        };
        
        const renderSummary = () => {
            const today = new Date().toISOString().split('T')[0];
            const todayLog = state.logs.find(log => log.date === today);

            document.getElementById('summary-calories').textContent = `${todayLog?.calorias || 0} kcal`;
            document.getElementById('summary-exercise').textContent = `${todayLog?.exercicios || 0} min`;
        };

        const updateAll = () => {
            renderMetrics();
            renderChart();
            renderHistory();
            renderSummary();
            saveState();
        };

        // --- MANIPULAÇÃO DE DADOS ---

        const addLog = (data) => {
            // Remove log existente para a mesma data, se houver
            state.logs = state.logs.filter(log => log.date !== data.date);
            
            const newLog = {
                id: Date.now().toString(),
                ...data
            };
            state.logs.push(newLog);
            updateAll();
        };

            }
        };

            state.profile = { ...state.profile, ...data };
            updateAll();
        };

            if (state.logs.length === 0) return null;
            return [...state.logs].sort((a, b) => new Date(b.date) - new Date(a.date))[0];

            for (let [key, value] of formData.entries()) {
                 const id = e.target.querySelector(`[name="${key}"]`)?.id || `log-${key}`;
